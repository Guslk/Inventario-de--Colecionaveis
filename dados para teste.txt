// registrar um usuario

Método: POST
URL: http://localhost:3000/api/usuarios
Body: JSON com nome, email, senha.
{
  "nome": "Usuario Teste",
  "email": "teste@exemplo.com",
  "senha": "123456"
}

// logar um usuario

Método: POST
URL: http://localhost:3000/api/usuarios/login
Body: JSON com email, senha.
{
  "email": "teste@exemplo.com",
  "senha": "123456"
}
